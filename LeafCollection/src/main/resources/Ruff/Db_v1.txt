-- =========================
-- BRANCH TABLE
-- =========================
CREATE TABLE branch (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


-- =========================
-- BRANCH MONTHLY RATE
-- Rate decided monthly per branch
-- =========================
CREATE TABLE branch_rate (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    branch_id BIGINT NOT NULL,
    month INT NOT NULL,
    year INT NOT NULL,
    rate_per_kg DECIMAL(10,2) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT unique_branch_month UNIQUE(branch_id, month, year),
    FOREIGN KEY (branch_id) REFERENCES branch(id)
);


-- =========================
-- PARTY TABLE
-- Leaf suppliers
-- =========================
CREATE TABLE party (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    branch_id BIGINT NOT NULL,
    opening_balance DECIMAL(12,2) DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (branch_id) REFERENCES branch(id)
);


-- =========================
-- DAILY LEAF COLLECTION ENTRY
-- Only quantity stored
-- =========================
CREATE TABLE leaf_entry (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    party_id BIGINT NOT NULL,
    entry_date DATE NOT NULL,
    quantity DECIMAL(12,2) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (party_id) REFERENCES party(id)
);


-- =========================
-- PAYMENT TABLE
-- Advance or settlement
-- =========================
CREATE TABLE payment (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    party_id BIGINT NOT NULL,
    payment_date DATE NOT NULL,
    amount DECIMAL(12,2) NOT NULL,
    payment_type VARCHAR(20),
    notes VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (party_id) REFERENCES party(id)
);


-- =========================
-- MONTHLY BILL TABLE
-- Finalized monthly calculation
-- =========================
CREATE TABLE monthly_bill (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    party_id BIGINT NOT NULL,
    month INT NOT NULL,
    year INT NOT NULL,
    total_quantity DECIMAL(12,2) NOT NULL,
    rate DECIMAL(10,2) NOT NULL,
    total_amount DECIMAL(12,2) NOT NULL,
    advance_paid DECIMAL(12,2) DEFAULT 0,
    balance DECIMAL(12,2) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT unique_party_month UNIQUE(party_id, month, year),
    FOREIGN KEY (party_id) REFERENCES party(id)
);
